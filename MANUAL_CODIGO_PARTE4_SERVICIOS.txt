```typescript
import { inject, Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { BehaviorSubject, Observable, tap } from 'rxjs';
import { Router } from '@angular/router';

interface LoginResponse {
  token: string;
  empleado: {
    id: number;
    nombre: string;
    email: string;
    esAdmin: boolean;
  };
}

@Injectable({
  providedIn: 'root'
})
export class AuthService {
  private http = inject(HttpClient);
  private router = inject(Router);

  private isLoggedSubject = new BehaviorSubject<boolean>(this.hasToken());
  isLogged$ = this.isLoggedSubject.asObservable();

  private isAdminSubject = new BehaviorSubject<boolean>(false);
  isAdmin$ = this.isAdminSubject.asObservable();

  private empleadoIdSubject = new BehaviorSubject<number | null>(null);
  empleadoId$ = this.empleadoIdSubject.asObservable();

  private apiUrl = '/api/auth';

  constructor() {
    this.loadTokenInfo();
  }

  login(email: string, password: string): Observable<LoginResponse> {
    return this.http.post<LoginResponse>(`${this.apiUrl}/login`, { email, password })
      .pipe(
        tap(response => this.setSession(response))
      );
  }

  private setSession(authResult: LoginResponse): void {
    localStorage.setItem('token', authResult.token);
    localStorage.setItem('esAdmin', String(authResult.empleado.esAdmin));
    localStorage.setItem('empleadoId', String(authResult.empleado.id));

    this.isLoggedSubject.next(true);
    this.isAdminSubject.next(authResult.empleado.esAdmin);
    this.empleadoIdSubject.next(authResult.empleado.id);
  }

  logout(): void {
    localStorage.removeItem('token');
    localStorage.removeItem('esAdmin');
    localStorage.removeItem('empleadoId');

    this.isLoggedSubject.next(false);
    this.isAdminSubject.next(false);
    this.empleadoIdSubject.next(null);
  }

  isLoggedIn(): boolean {
    return this.hasToken();
  }

  isAdmin(): boolean {
    return localStorage.getItem('esAdmin') === 'true';
  }

  getEmpleadoId(): number | null {
    const id = localStorage.getItem('empleadoId');
    return id ? +id : null;
  }

  getToken(): string | null {
    return localStorage.getItem('token');
  }

  private hasToken(): boolean {
    return !!localStorage.getItem('token');
  }

  private loadTokenInfo(): void {
    if (this.hasToken()) {
      this.isLoggedSubject.next(true);
      this.isAdminSubject.next(this.isAdmin());
      this.empleadoIdSubject.next(this.getEmpleadoId());
    }
  }
}
```

```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface Empleado {
  id: number;
  dni: string;
  nombre: string;
  apellidos: string;
  fechaNacimiento: string;
  email: string;
  password: string;
  telefono: string;
  esAdmin: boolean;
}

@Injectable({
  providedIn: 'root'
})
export class EmpleadosService {
  private http = inject(HttpClient);
  private apiUrl = '/api/empleados';

  getEmpleados(): Observable<Empleado[]> {
    return this.http.get<Empleado[]>(this.apiUrl);
  }

  getEmpleado(id: number): Observable<Empleado> {
    return this.http.get<Empleado>(`${this.apiUrl}/${id}`);
  }

  crearEmpleado(empleado: Empleado): Observable<Empleado> {
    return this.http.post<Empleado>(this.apiUrl, empleado);
  }

  actualizarEmpleado(id: number, empleado: Empleado): Observable<any> {
    return this.http.put(`${this.apiUrl}/${id}`, empleado);
  }

  eliminarEmpleado(id: number): Observable<any> {
    return this.http.delete(`${this.apiUrl}/${id}`);
  }
}
```

```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface Fichaje {
  id: number;
  empleadoId: number;
  fecha: string;
  horaEntrada: string;
  horaSalida: string | null;
}

export interface Incidencia {
  id: number;
  fichajeId: number;
  descripcion: string;
  fechaHora: string;
}

export interface FichajeConIncidencia {
  id: number;
  empleadoId: number;
  fecha: string;
  horaEntrada: string;
  horaSalida: string | null;
  incidencias: Incidencia[];
}

@Injectable({
  providedIn: 'root'
})
export class FichajesService {
  private http = inject(HttpClient);
  private apiUrl = '/api/fichajes';

  registrarEntrada(empleadoId: number): Observable<Fichaje> {
    return this.http.post<Fichaje>(`${this.apiUrl}/entrada`, { empleadoId });
  }

  registrarSalida(empleadoId: number): Observable<Fichaje> {
    return this.http.post<Fichaje>(`${this.apiUrl}/salida`, { empleadoId });
  }

  registrarIncidencia(fichajeId: number, descripcion: string): Observable<Incidencia> {
    return this.http.post<Incidencia>(`${this.apiUrl}/${fichajeId}/incidencias`, { descripcion });
  }

  obtenerFichajesEmpleado(empleadoId: number): Observable<FichajeConIncidencia[]> {
    return this.http.get<FichajeConIncidencia[]>(`${this.apiUrl}/empleado/${empleadoId}`);
  }

  obtenerFichajesPorFecha(empleadoId: number, fechaInicio: string, fechaFin: string): Observable<FichajeConIncidencia[]> {
    return this.http.get<FichajeConIncidencia[]>(
      `${this.apiUrl}/empleado/${empleadoId}/rango?fechaInicio=${fechaInicio}&fechaFin=${fechaFin}`
    );
  }
}
```

```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface Ausencia {
  id: number;
  empleadoId: number;
  empleadoNombre?: string;
  fechaInicio: string;
  fechaFin: string;
  motivo: string;
  estado: 'Pendiente' | 'Aprobada' | 'Rechazada';
}

@Injectable({
  providedIn: 'root'
})
export class AusenciasService {
  private http = inject(HttpClient);
  private apiUrl = '/api/ausencias';

  crearAusencia(ausencia: Omit<Ausencia, 'id' | 'estado'>): Observable<Ausencia> {
    return this.http.post<Ausencia>(this.apiUrl, ausencia);
  }

  obtenerAusenciasEmpleado(empleadoId: number): Observable<Ausencia[]> {
    return this.http.get<Ausencia[]>(`${this.apiUrl}/empleado/${empleadoId}`);
  }

  obtenerTodasAusencias(): Observable<Ausencia[]> {
    return this.http.get<Ausencia[]>(this.apiUrl);
  }

  aprobarAusencia(id: number): Observable<any> {
    return this.http.put(`${this.apiUrl}/${id}/aprobar`, {});
  }

  rechazarAusencia(id: number): Observable<any> {
    return this.http.put(`${this.apiUrl}/${id}/rechazar`, {});
  }
}
```
